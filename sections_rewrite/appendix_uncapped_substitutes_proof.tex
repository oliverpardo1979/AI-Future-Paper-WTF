\begin{proof}[Proof of Proposition~\ref{prop:rewrite-uncapped-substitutes-explosion}]
\phantomsection
\label{proof:rewrite-uncapped-substitutes-explosion}
Suppose, for contradiction, that a positive finite path exists for all
$t\geq t_0$ and satisfies the stated static conditions, feasibility, and
equation~\eqref{eq:rewrite-uncapped-investment-floors}. Within this proof,
write $z=Y/K$, $s=s_X$, and $\theta=(1-\alpha)/\alpha>0$.

\textit{Step 1. The return bound from Section~\ref{subsec:rewrite-return-threshold}.}
Let $e(s)=(1-s)/\sigma+\alpha s$, as in
equation~\eqref{eq:rewrite-inverse-demand-elasticity}. The pricing,
monopoly, and service equations~\eqref{eq:rewrite-ai-price},
\eqref{eq:rewrite-monopoly-foc}, and~\eqref{eq:rewrite-ai-service} give
$U/Y=(1-\alpha)s[1-e(s)]$ and $X/Y=B(1-\alpha)s[1-e(s)]$.
The CES and its elasticity in equations~\eqref{eq:rewrite-ces}
and~\eqref{eq:rewrite-sx} give
$Z/X=(\omega_X/s)^{\sigma/(\sigma-1)}$. Substituting both ratios into
production, equation~\eqref{eq:rewrite-output}, yields
\begin{equation}
 z=\left[
 \omega_X^{\sigma/(\sigma-1)}B(1-\alpha)[1-e(s)]
 s^{-1/(\sigma-1)}\right]^\theta.
 \label{eq:rewrite-uncapped-explosion-static}
\end{equation}
Define $f(s)=[1-e(s)]s^{-1/(\sigma-1)}$. Its logarithmic derivative is
\[
 \frac{\mathrm d\log f}{\mathrm d\log s}
 =\frac{(\sigma-2)(1-\alpha\sigma)s-(\sigma-1)}
 {\sigma(\sigma-1)[1-e(s)]}.
\]
The denominator is positive. The numerator is affine in $s$ and equals
$-(\sigma-1)<0$ at $s=0$ and
$-[1-\alpha+\alpha(\sigma-1)^2]<0$ at $s=1$. Thus $f$ is strictly
decreasing on $(0,1)$, with $f(1)=1-\alpha$. Set
\[
 a_X=\left[(1-\alpha)^2\omega_X^{\sigma/(\sigma-1)}\right]^\theta>0.
\]
Equation~\eqref{eq:rewrite-uncapped-explosion-static} then implies
\begin{equation}
 z\geq a_XB^\theta,\qquad
 r=\alpha z-\delta\geq\alpha a_XB^\theta-\delta=\mathcal R(B),
 \label{eq:rewrite-uncapped-output-capital-bound}
\end{equation}
using the return definition~\eqref{eq:rewrite-ai-terminal-ratio}.
This proves equation~\eqref{eq:rewrite-uncapped-dated-return-bound}.
The inequalities are strict for $0<s_X<1$, and equality is attained only
in the limiting static allocation $s_X\to1$. This step uses neither
investment bounds nor an assumption about the long-run behavior of $B$.

\textit{Step 2. Persistent investment implies unbounded AI efficiency.}
Since $\sigma>1$, the CES also gives
\[
 Y\geq c_LK^\alpha(AL)^{1-\alpha},\qquad
 c_L=\omega_L^{\sigma(1-\alpha)/(\sigma-1)}>0.
\]
Effective labor is positive and nondecreasing. The gross investment bound
therefore implies
\[
 \dot K\geq\iota c_LK^\alpha[AL(t_0)]^{1-\alpha}-\delta K.
\]
For $\delta>0$, capital cannot cross below the positive constant
\[
 K_{\min}=\min\left\{K(t_0),
 \left(\frac{\iota c_L[AL(t_0)]^{1-\alpha}}{\delta}
 \right)^{1/(1-\alpha)}\right\}.
\]
For $\delta=0$, take $K_{\min}=K(t_0)$. Consequently
$Y\geq Y_{\min}:=c_LK_{\min}^{\alpha}[AL(t_0)]^{1-\alpha}>0$
and $M\geq\mu Y_{\min}$. Integrating the research
law~\eqref{eq:rewrite-uncapped-unit-law} gives
\begin{equation}
 B_t^{1-\eta}\geq B_{t_0}^{1-\eta}
 +(1-\eta)\chi(\mu Y_{\min})^\eta(t-t_0).
 \label{eq:rewrite-uncapped-persistent-research}
\end{equation}
Thus $B$ would tend to infinity if the path were global. This conclusion
is derived, not assumed. Nondecreasing $AL$ suffices for this step,
including constant effective labor.

\textit{Step 3. Capital and research imply a superlinear growth bound.}
Equations~\eqref{eq:rewrite-uncapped-output-capital-bound}
and~\eqref{eq:rewrite-uncapped-persistent-research} imply $z\to\infty$.
Hence there is a finite $t_1$ after which $\delta/z\leq\iota/2$ and
\begin{equation}
 \dot K\geq\frac{\iota}{2}Y,\qquad
 \chi\mu^\eta B^\eta Y^\eta\leq\dot B\leq\chi B^\eta Y^\eta.
 \label{eq:rewrite-uncapped-stock-growth-bounds}
\end{equation}
The upper research bound uses $M\leq Y$, which follows from the resource
constraint, nonnegative consumption and inference compute, and positive
gross capital investment. The factor $1/2$ is a convenient proof margin,
not an additional restriction.

Since $\dot B>0$, use $B$ as the independent variable along the proposed
path. Divide the lower bound on $\dot K$ by the \emph{upper} bound on
$\dot B$ to obtain
\begin{align}
 \frac{\mathrm dK^\eta}{\mathrm dB}
 &=\frac{\eta K^{\eta-1}\dot K}{\dot B}
 \geq\frac{\eta\iota}{2\chi}B^{-\eta}z^{1-\eta}\nonumber\\
 &\geq\frac{\eta\iota a_X^{1-\eta}}{2\chi}
 B^{(1-\alpha-\eta)/\alpha}.
 \label{eq:rewrite-uncapped-capital-feedback}
\end{align}
Because $(1-\eta)/\alpha>0$, integration yields
\[
 K^\eta\geq K(t_1)^\eta+
 c_K\left[B^{(1-\eta)/\alpha}-B(t_1)^{(1-\eta)/\alpha}\right],
 \qquad
 c_K=\frac{\eta\iota\alpha a_X^{1-\eta}}{2\chi(1-\eta)}>0.
\]
For sufficiently large $B$, this implies
$K^\eta\geq(c_K/2)B^{(1-\eta)/\alpha}$. Substituting into the
\emph{lower} bound on $\dot B$ in
equation~\eqref{eq:rewrite-uncapped-stock-growth-bounds} gives
\begin{equation}
 \dot B\geq c_BB^{1/\alpha},\qquad
 c_B=\frac{\chi\mu^\eta a_X^\eta c_K}{2}>0.
 \label{eq:rewrite-uncapped-superlinear-feedback}
\end{equation}
Indeed, the exponent is
$\eta+\eta(1-\alpha)/\alpha+(1-\eta)/\alpha=1/\alpha>1$.
It is greater than one for every $0<\eta<1$, on either side of
$\eta=\alpha$.

\textit{Step 4. Contradiction with a globally finite trajectory.}
For a sufficiently late $t_2$, equation~\eqref{eq:rewrite-uncapped-superlinear-feedback}
implies
\[
 \frac{\mathrm d}{\mathrm dt}B^{1-1/\alpha}
 \leq-(1/\alpha-1)c_B,
 \qquad
 B_t^{1-1/\alpha}\leq
 B_{t_2}^{1-1/\alpha}-(1/\alpha-1)c_B(t-t_2).
\]
The right side reaches zero at the finite date
\[
 t_2+\frac{B_{t_2}^{1-1/\alpha}}{(1/\alpha-1)c_B}.
\]
The left side must remain strictly positive whenever $B$ is positive
and finite. This contradiction is the scalar comparison underlying
Osgood's explosion criterion; see
\citet[Lemma 2.2 and Theorem 3.1]{ceballosetal2011}.
No globally finite feasible path with the two investment bounds exists.
Since every equilibrium must be feasible and have finite allocations at
every finite date, no equilibrium can satisfy those bounds either.
\end{proof}
